merge intervals


import java.util.Arrays; // Импортируем Arrays, чтобы использовать сортировку массивов и другие полезные методы.
import java.util.ArrayList; // Импортируем ArrayList — динамический список, размер которого может увеличиваться.
import java.util.List; // Импортируем интерфейс List, через который будем работать со списком интервалов.

class Solution { 

    public int[][] merge(int[][] intervals) { // Создаём метод merge, который принимает двумерный массив интервалов и возвращает двумерный массив.

        Arrays.sort(intervals, (a, b) -> Integer.compare(a[0], b[0])); // Сортируем интервалы по первому числу: по возрастанию начала интервала.

        List<int[]> result = new ArrayList<>();  
        for (int[] interval : intervals) { // Перебираем каждый интервал в отсортированном массиве.

            if (result.isEmpty()) { // Проверяем, пуст ли результат: если да, то пока не с чем сравнивать текущий интервал.

                result.add(interval); // Добавляем первый интервал в результат.

            } else { // Если список уже содержит хотя бы один интервал, сравниваем текущий интервал с последним в результате.

                int[] last = result.get(result.size() - 1); // Получаем последний интервал из result: size() - 1 — индекс последнего элемента.

                if (interval[0] <= last[1]) { // Проверяем, пересекаются ли интервалы: начало текущего <= конец последнего.

                    last[1] = Math.max(last[1], interval[1]); // Объединяем интервалы: сохраняем самое большое значение конца из двух интервалов.

                } else { // Если начало текущего интервала больше конца последнего, интервалы не пересекаются.

                    result.add(interval); // Добавляем текущий интервал в результат как отдельный интервал.

                } // Заканчиваем проверку на пересечение.

            } // Заканчиваем проверку на пустой список.

        } // Заканчиваем цикл, потому что все интервалы обработаны.

        return result.toArray(new int[result.size()][]); // Преобразуем List<int[]> в требуемый двумерный массив int[][] и возвращаем результат.

    } // Заканчиваем метод merge.

}
